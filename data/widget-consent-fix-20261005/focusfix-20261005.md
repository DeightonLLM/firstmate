# FOCUSFIX-20261005 KEYBOARD-RESIZE FIX

Appended guarded IIFE `focusfix-20261005` after `widgetfix-20261005`, `consentfix-20261005`, and `hdsizingfix-20261005`. It tracks input focus plus `window.visualViewport` resize/scroll; updates are requestAnimationFrame scheduled. While focused, it sizes the chat panel to the visual viewport, fixes the composer at the calculated visual viewport bottom with safe-area inset subtraction, and retains 100dvh/100svh fallback sizing. It restores prior page scrollY on blur and uses `overscroll-behavior: contain`. When focused, an existing viewport meta receives `interactive-widget=resizes-content` only if no interactive-widget setting exists; existing directives are preserved. Browsers without visualViewport retain prior widget behavior.

```js
/* focusfix-20261005 — visualViewport keyboard-aware composer positioning */
(function () {
  "use strict";
  var ID = "focusfix-20261005";
  if (document.getElementById(ID)) return;
  try {
    var marker = document.createElement("meta");
    marker.id = ID;
    (document.head || document.documentElement).appendChild(marker);
    var vv = window.visualViewport;
    var active = false;
    var savedScrollY = 0;
    var frame = 0;
    var style = document.createElement("style");
    style.id = ID + "-style";
    style.textContent = "#anything-llm-chat.focusfix-keyboard,#anything-llm-embed-chat-container.focusfix-keyboard{height:var(--focusfix-vv-height,100dvh)!important;max-height:var(--focusfix-vv-height,100svh)!important;overscroll-behavior:contain!important}#anything-llm-chat.focusfix-keyboard form,#anything-llm-embed-chat-container.focusfix-keyboard form{position:fixed!important;left:12px!important;right:12px!important;bottom:var(--focusfix-vv-bottom,0px)!important;z-index:20!important}";
    (document.head || document.documentElement).appendChild(style);

    function addInteractiveWidget() {
      var meta = document.querySelector('meta[name="viewport"]');
      if (!meta || /(?:^|[,\s])interactive-widget\s*=/.test(meta.content || "")) return;
      meta.setAttribute("content", (meta.content || "width=device-width, initial-scale=1") + ", interactive-widget=resizes-content");
    }
    function applyViewport() {
      frame = 0;
      if (!active || !vv) return;
      var chat = document.getElementById("anything-llm-chat");
      var embed = document.getElementById("anything-llm-embed-chat-container");
      var height = Math.max(0, vv.height);
      var bottom = Math.max(0, height - vv.offsetTop);
      [chat, embed].forEach(function (el) {
        if (!el) return;
        el.classList.add("focusfix-keyboard");
        el.style.setProperty("--focusfix-vv-height", height + "px");
        el.style.setProperty("--focusfix-vv-bottom", "max(0px, calc(" + bottom + "px - env(safe-area-inset-bottom, 0px)))");
      });
    }
    function schedule() {
      if (!frame) frame = window.requestAnimationFrame(applyViewport);
    }
    function focused(event) {
      var target = event.target;
      if (!target || !target.matches || !target.matches("#message-input, #anything-llm-chat form textarea, #anything-llm-chat form input")) return;
      if (!active) savedScrollY = window.scrollY || window.pageYOffset || 0;
      active = true;
      addInteractiveWidget();
      schedule();
    }
    function blurred(event) {
      var next = event.relatedTarget;
      if (next && next.matches && next.matches("#message-input, #anything-llm-chat form textarea, #anything-llm-chat form input")) return;
      active = false;
      if (frame) { window.cancelAnimationFrame(frame); frame = 0; }
      document.querySelectorAll(".focusfix-keyboard").forEach(function (el) {
        el.classList.remove("focusfix-keyboard");
        el.style.removeProperty("--focusfix-vv-height");
        el.style.removeProperty("--focusfix-vv-bottom");
      });
      window.requestAnimationFrame(function () { window.scrollTo(0, savedScrollY); });
    }
    document.addEventListener("focusin", focused, true);
    document.addEventListener("focusout", blurred, true);
    if (vv) {
      vv.addEventListener("resize", schedule, { passive: true });
      vv.addEventListener("scroll", schedule, { passive: true });
    }
    if (!vv) return; /* existing widgetfix rules remain the graceful fallback */
  } catch (_e) { /* never break the widget */ }
})();
```
